' 对齐常量定义
Public Const ALIGN_LEFT = 1
Public Const ALIGN_CENTER_H = 2
Public Const ALIGN_RIGHT = 3
Public Const ALIGN_TOP = 4
Public Const ALIGN_CENTER_V = 5
Public Const ALIGN_BOTTOM = 6
Public Const ALIGN_CENTER = 7

' 对齐基准常量
Public Const BASE_PAGE = 1          ' 以页面为基准
Public Const BASE_TOP_OBJECT = 2    ' 以顶层对象为基准（显示在最上面的）
Public Const BASE_BOTTOM_OBJECT = 3 ' 以底层对象为基准（显示在最下面的）

' ==================== 核心对齐函数 ====================

' 对象对齐函数（alignType:1-7, baseType:1-3）
Sub QuickAlignToPage(ByVal alignType As Integer, ByVal baseType As Integer)
    Dim s As shape
    Dim pageW As Double
    Dim pageH As Double
    Dim newX As Double
    Dim newY As Double
    Dim refShape As shape
    Dim refX As Double
    Dim refY As Double
    Dim refW As Double
    Dim refH As Double
    Dim isSkip As Boolean
    
    ' 检查文档
    If ActiveDocument Is Nothing Then
        Exit Sub
    End If
    
    ' 检查选中对象
    If ActiveSelection.Shapes.count = 0 Then
        MsgBox "请先选择对象！"
        Exit Sub
    End If
    
    ' 获取参考基准
    If baseType = BASE_PAGE Then
        ' 页面对齐：使用页面尺寸
        pageW = ActivePage.SizeWidth
        pageH = ActivePage.SizeHeight
    Else
        ' 对象对齐：获取参考对象（顶层或底层）
        If baseType = BASE_TOP_OBJECT Then
            Set refShape = GetTopmostShape()
        ElseIf baseType = BASE_BOTTOM_OBJECT Then
            Set refShape = GetBottommostShape()
        End If
        
        ' 检查参考对象
        If refShape Is Nothing Then
            MsgBox "无法获取参考对象！"
            Exit Sub
        End If
        
        ' 获取参考对象的位置和尺寸
        refX = refShape.PositionX
        refY = refShape.PositionY
        refW = refShape.SizeWidth
        refH = refShape.SizeHeight
    End If
    
    ' 开始事务
    ActiveDocument.BeginCommandGroup "对齐对象"
    Application.Optimization = True
    
    ' 遍历所有选中的形状
    For Each s In ActiveSelection.Shapes
        isSkip = False
        
        ' 跳过参考对象本身（对象对齐时）
        If baseType <> BASE_PAGE Then
            If s Is refShape Then
                isSkip = True
            End If
        End If
        
        If Not isSkip Then
            ' 根据基准类型计算新位置
            If baseType = BASE_PAGE Then
                ' 页面对齐
                Select Case alignType
                    Case ALIGN_LEFT
                        newX = 0
                        newY = s.PositionY
                    Case ALIGN_CENTER_H
                        newX = (pageW - s.SizeWidth) / 2
                        newY = s.PositionY
                    Case ALIGN_RIGHT
                        newX = pageW - s.SizeWidth
                        newY = s.PositionY
                    Case ALIGN_TOP
                        newX = s.PositionX
                        newY = pageH
                    Case ALIGN_CENTER_V
                        newX = s.PositionX
                        newY = (pageH + s.SizeHeight) / 2
                    Case ALIGN_BOTTOM
                        newX = s.PositionX
                        newY = pageH - (pageH - s.SizeHeight)
                    Case ALIGN_CENTER
                        newX = (pageW - s.SizeWidth) / 2
                        newY = (pageH + s.SizeHeight) / 2
                End Select
            Else
                ' 对象对齐
                Select Case alignType
                    Case ALIGN_LEFT
                        newX = refX
                        newY = s.PositionY
                    Case ALIGN_CENTER_H
                        newX = refX + (refW - s.SizeWidth) / 2
                        newY = s.PositionY
                    Case ALIGN_RIGHT
                        newX = refX + refW - s.SizeWidth
                        newY = s.PositionY
                    Case ALIGN_TOP
                        newX = s.PositionX
                        newY = refY
                    Case ALIGN_CENTER_V
                        newX = s.PositionX
                        newY = refY + (s.SizeHeight - refH) / 2
                    Case ALIGN_BOTTOM
                        newX = s.PositionX
                        newY = refY + (s.SizeHeight - refH)
                    Case ALIGN_CENTER
                        newX = refX + (refW - s.SizeWidth) / 2
                        newY = refY + (s.SizeHeight - refH) / 2
                End Select
            End If
            
            ' 设置新位置
            s.SetPosition newX, newY
        End If
    Next s
    
    ' 恢复设置
    Application.Optimization = False
    ActiveDocument.EndCommandGroup
    Application.Refresh
End Sub

' 获取选中最顶层的对象（显示在最上面，ZOrder最大）
Function GetTopmostShape() As shape
    Dim s As shape
    Dim topmost As shape
    Dim maxZOrder As Long
    
    ' 初始化
    Set topmost = Nothing
    maxZOrder = 999999
    
    ' 检查选中对象
    If ActiveSelection.Shapes.count = 0 Then
        Set GetTopmostShape = Nothing
        Exit Function
    End If
    
    ' 遍历查找最大 ZOrder（最顶层）
    For Each s In ActiveSelection.Shapes
        If s.ZOrder < maxZOrder Then
            maxZOrder = s.ZOrder
            Set topmost = s
        End If
    Next s
    
    Set GetTopmostShape = topmost
End Function

' 获取选中最低层的对象（显示在最下面，ZOrder最小）
Function GetBottommostShape() As shape
    Dim s As shape
    Dim bottommost As shape
    Dim minZOrder As Long
    
    ' 初始化
    Set bottommost = Nothing
    minZOrder = -1
    
    ' 检查选中对象
    If ActiveSelection.Shapes.count = 0 Then
        Set GetBottommostShape = Nothing
        Exit Function
    End If
    
    ' 遍历查找最小 ZOrder（最底层）
    For Each s In ActiveSelection.Shapes
        If s.ZOrder > minZOrder Then
            minZOrder = s.ZOrder
            Set bottommost = s
        End If
    Next s
    
    Set GetBottommostShape = bottommost
End Function
