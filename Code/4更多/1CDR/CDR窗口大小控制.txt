Private Declare PtrSafe Function ShowWindow Lib "user32" _
    (ByVal hwnd As LongPtr, ByVal nCmdShow As Long) As Long

Private Declare PtrSafe Function GetForegroundWindow Lib "user32" () As LongPtr

Private Declare PtrSafe Function GetWindowTextLength Lib "user32" Alias "GetWindowTextLengthA" _

Private Declare PtrSafe Function GetWindowText Lib "user32" Alias "GetWindowTextA" _
    (ByVal hwnd As LongPtr, ByVal lpString As String, ByVal cch As Long) As Long

Private Declare PtrSafe Function EnumWindows Lib "user32" _
    (ByVal lpEnumFunc As LongPtr, ByVal lParam As LongPtr) As Long

Private Declare PtrSafe Function GetClassName Lib "user32" Alias "GetClassNameA" _
    (ByVal hwnd As LongPtr, ByVal lpClassName As String, ByVal nMaxCount As Long) As Long

' 窗口状态常量
Private Const SW_MINIMIZE As Long = 6
Private Const SW_MAXIMIZE As Long = 3
Private Const SW_RESTORE As Long = 9

' 全局变量用于枚举窗口
Private g_hWndCDR As LongPtr
Private g_strWindowTitle As String

' 获取CorelDRAW主窗口句柄
Private Function GetCDRWindowHandle() As LongPtr
    Dim hwnd As LongPtr
    Dim titleLen As Long
    Dim title As String
    Dim classLen As Long
    Dim vClassName As Variant
    
    ' 方法1：尝试通过已知类名查找
    Dim classNames As Variant
    classNames = Array("CorelDRAW", "CorelDRAW.Application", "CorelDRW", "CorelDRAW2024", "CorelDRAW 2024")
    
    For Each vClassName In classNames
        hwnd = FindWindow(CStr(vClassName), vbNullString)
        If hwnd <> 0 Then
            GetCDRWindowHandle = hwnd
            Exit Function
        End If
    Next
    
    ' 方法2：通过窗口标题查找（包含"CorelDRAW"的窗口）
    g_hWndCDR = 0
    g_strWindowTitle = ""
    EnumWindows AddressOf EnumWindowProc, 0
    
    If g_hWndCDR <> 0 Then
        GetCDRWindowHandle = g_hWndCDR
        Exit Function
    End If
    
    ' 方法3：获取前台窗口（如果CDR是活动窗口）
    hwnd = GetForegroundWindow()
    If hwnd <> 0 Then
        titleLen = GetWindowTextLength(hwnd) + 1
        title = String(titleLen, Chr(0))
        GetWindowText hwnd, title, titleLen
        title = Left(title, titleLen - 1)
        
        If InStr(1, title, "CorelDRAW", vbTextCompare) > 0 Then
            GetCDRWindowHandle = hwnd
            Exit Function
        End If
    End If
    
    GetCDRWindowHandle = 0
End Function

' 枚举窗口回调函数
Private Function EnumWindowProc(ByVal hwnd As LongPtr, ByVal lParam As LongPtr) As Long
    Dim titleLen As Long
    Dim title As String
    Dim className As String
    Dim classLen As Long
    
    ' 获取窗口标题
    titleLen = GetWindowTextLength(hwnd) + 1
    If titleLen > 1 Then
        title = String(titleLen, Chr(0))
        GetWindowText hwnd, title, titleLen
        title = Left(title, titleLen - 1)
        
        ' 检查标题是否包含CorelDRAW
        If InStr(1, title, "CorelDRAW", vbTextCompare) > 0 Then
            g_hWndCDR = hwnd
            EnumWindowProc = 0 ' 停止枚举
            Exit Function
        End If
    End If
    
    EnumWindowProc = 1 ' 继续枚举
End Function

' 最小化CorelDRAW 2024主窗口
Sub MinimizeCorelDRAWWindow()
    Dim hWndCDR As LongPtr
    
    hWndCDR = GetCDRWindowHandle()
    
    If hWndCDR <> 0 Then
        ShowWindow hWndCDR, SW_MINIMIZE
    Else
        MsgBox "未找到CorelDRAW窗口", vbExclamation
    End If
End Sub

' 最大化CorelDRAW 2024主窗口
Sub MaximizeCorelDRAWWindow()
    Dim hWndCDR As LongPtr
    
    hWndCDR = GetCDRWindowHandle()
    
    If hWndCDR <> 0 Then
        ShowWindow hWndCDR, SW_MAXIMIZE
    Else
        MsgBox "未找到CorelDRAW窗口", vbExclamation
    End If
End Sub

' 还原CorelDRAW 2024主窗口
Sub RestoreCorelDRAWWindow()
    Dim hWndCDR As LongPtr
    
    hWndCDR = GetCDRWindowHandle()
    
    If hWndCDR <> 0 Then
        ShowWindow hWndCDR, SW_RESTORE
    Else
        MsgBox "未找到CorelDRAW窗口", vbExclamation
    End If
End Sub
