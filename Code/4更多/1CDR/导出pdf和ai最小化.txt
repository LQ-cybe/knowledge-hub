' 全局变量：供枚举窗口回调函数使用
#If VBA7 Then
    Private targetPID As LongPtr
    Private cdrHwnd As LongPtr
#Else
    Private targetPID As Long
    Private cdrHwnd As Long
#End If


' 核心函数：最小化当前CorelDRAW窗口
' 返回值：True=成功最小化，False=失败
Private Function MinimizeCDRWindow() As Boolean
    MinimizeCDRWindow = False ' 默认为失败
    cdrHwnd = 0 ' 重置窗口句柄
    
    ' 获取当前CorelDRAW进程ID
    targetPID = GetCurrentProcessId()
    
    ' 枚举所有窗口，找到当前进程的可见主窗口
    EnumWindows AddressOf EnumWindowsProc, 0
    
    ' 执行最小化并返回结果
    If cdrHwnd <> 0 Then
        ShowWindow cdrHwnd, 6 ' 6=SW_MINIMIZE
        MinimizeCDRWindow = True
    End If
End Function


' 窗口枚举回调函数（内部使用）
#If VBA7 Then
    Private Function EnumWindowsProc(ByVal hwnd As LongPtr, ByVal lParam As LongPtr) As LongPtr
#Else
    Private Function EnumWindowsProc(ByVal hwnd As Long, ByVal lParam As Long) As Long
#End If
    Dim procId As Variant
    GetWindowThreadProcessId hwnd, procId ' 获取窗口所属进程ID
    
    ' 匹配当前进程且窗口可见
    If procId = targetPID And IsWindowVisible(hwnd) <> 0 Then
        cdrHwnd = hwnd ' 记录窗口句柄
        EnumWindowsProc = 0 ' 停止枚举
    Else
        EnumWindowsProc = 1 ' 继续枚举
    End If
End Function


' 主导出函数
Sub ExportToAI()
    Dim expopt As StructExportOptions
    Dim expflt As ExportFilter
    Dim docName As String
    Dim desktopPath As String
    Dim exportPath As String

    ' 检查活动文档
    If ActiveDocument Is Nothing Then
        MsgBox "请先打开一个文档", vbExclamation
        Exit Sub
    End If

    ' 获取文档名和导出路径
    docName = Left(ActiveDocument.Name, InStrRev(ActiveDocument.Name, ".") - 1)
    desktopPath = Environ("USERPROFILE") & "\Desktop\"
    exportPath = desktopPath & docName & ".ai"

    ' 执行导出
    Set expopt = CreateStructExportOptions
    expopt.UseColorProfile = True
    Set expflt = ActiveDocument.ExportEx(exportPath, cdrAI, cdrAllPages, expopt)
    
    With expflt
        .Version = 9
        .TextAsCurves = False
        .PreserveTransparency = True
        .Finish
    End With

    ' 调用封装的最小化函数（灵活调用点）
    If Not MinimizeCDRWindow() Then
        MsgBox "已导出到：" & vbCrLf & exportPath & vbCrLf & "但未找到窗口进行最小化", vbInformation
    End If
End Sub
