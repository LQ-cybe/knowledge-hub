'加载指定目录插件
Sub LoadAddin()
    Dim addinPath As String
    Dim addin As addin   ' 改这里：AddIn 而不是 addin
    
    ' 指定xlam文件路径
    addinPath = "C:\Users\LQ\AppData\Roaming\Microsoft\AddIns\我的插件.xlam"
    
    ' 检查文件是否存在
    If Dir(addinPath) = "" Then
        MsgBox "找不到文件：" & addinPath, vbExclamation
        Exit Sub
    End If
    
    ' 加载插件
    On Error Resume Next
    Set addin = Application.AddIns.Add(addinPath)
    addin.Installed = True
    On Error GoTo 0
    
    If addin.Installed Then
        MsgBox "插件加载成功！", vbInformation
    Else
        MsgBox "插件加载失败！", vbExclamation
    End If
End Sub
