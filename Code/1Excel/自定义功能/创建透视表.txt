'创建透视表
Sub CreatePivot_Basic_Fixed()
    Dim wsSrc As Worksheet, wsDst As Worksheet
    Dim src As Range, pCache As PivotCache, pTable As PivotTable

    '=== 1. 明确指定工作表，防止多张同名工作表或隐藏表造成 CurrentRegion 取错 ===
    Set wsSrc = ThisWorkbook.Worksheets("Sheet1")
    Set wsDst = ThisWorkbook.Worksheets("Sheet2")

    '=== 2. 确保 Sheet2 的 B3 开始区域是干净的，否则 CreatePivotTable 会报错 ===
    wsDst.Cells.Clear

    '=== 3. 取数据区域：用显式列范围，避免 CurrentRegion 把空行/空列也圈进来 ===
    Dim lastRow As Long, lastCol As Long
    With wsSrc
        lastRow = .Cells(.Rows.Count, "A").End(xlUp).Row
        lastCol = .Cells(1, .Columns.Count).End(xlToLeft).Column
        Set src = .Range(.Cells(1, 1), .Cells(lastRow, lastCol))
    End With

    '=== 4. 如果之前已经建过同名透视表，先删除，否则会提示“名称已存在” ===
    On Error Resume Next
    wsDst.PivotTables("ptSales").TableRange2.Clear
    On Error GoTo 0

    '=== 5. 建立缓存并建表（Excel 2016+ 建议用 xlPivotTableVersion15，向下兼容 ===）
    Set pCache = ThisWorkbook.PivotCaches.Create( _
                     SourceType:=xlDatabase, _
                     SourceData:=src.Address(True, True, xlA1, True))    '带工作表绝对地址

    Set pTable = pCache.CreatePivotTable( _
                     TableDestination:=wsDst.Range("B3"), _
                     TableName:="ptSales")

    '=== 6. 添加字段前先把行/列/页字段全部先清掉（可选，保险） ===
    pTable.ClearTable

    '=== 7. 添加字段（写法更健壮：显式指定函数、名称） ===
    With pTable
        ' 行字段
        .PivotFields("产品").Orientation = xlRowField
        ' 列字段
        .PivotFields("地区").Orientation = xlColumnField
        ' 值字段（同时设置汇总方式和显示名称）
        With .PivotFields("销售额")
            .Orientation = xlDataField
            .Function = xlSum
            .Name = "销售额 汇总"
        End With
    End With
End Sub
