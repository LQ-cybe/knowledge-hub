'统计C列单元格中含TZH开头的字符串的行数和此字符串所在的行数，并写入在D、E列
Sub CountTZHLine_WithRowNum1()
    Dim rng As Range, cell As Range
    Dim arrLines As Variant, i As Long, cnt As Long
    Dim sht As Worksheet, strRowNums As String
    
    Set sht = ActiveSheet
    Set rng = sht.Range("C1:C" & sht.Cells(sht.Rows.Count, "C").End(xlUp).Row)
    
    Application.ScreenUpdating = False
    For Each cell In rng
        cnt = 0
        strRowNums = ""
        If Len(cell.Value) > 0 Then
            arrLines = Split(cell.Value, vbLf)
            For i = LBound(arrLines) To UBound(arrLines)
                If Left(Trim(arrLines(i)), 3) = "TZH" Then
                    cnt = cnt + 1
                    ' 拼接行号（i+1是因为数组从0开始，实际行号从1计数）
                    strRowNums = strRowNums & IIf(cnt > 1, vbLf, "") & (i + 1)
                End If
            Next i
        End If
        cell.Offset(0, 1).Value = cnt   ' D列：TZH开头总行数
        cell.Offset(0, 2).Value = strRowNums ' E列：TZH所在行号（多行）
    Next cell
    Application.ScreenUpdating = True
    MsgBox "统计完成！"
End Sub
