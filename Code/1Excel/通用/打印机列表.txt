'获取打印机列表
Sub ListAllPrintersUsingWMI()
    Dim ws As Worksheet
    Dim objWMIService As Object
    Dim colPrinters As Object
    Dim objPrinter As Object
    Dim i As Integer
    
    ' 创建一个新工作表
    On Error Resume Next
    Application.DisplayAlerts = False
    Worksheets("打印机列表").Delete
    Application.DisplayAlerts = True
    On Error GoTo 0
    
    Set ws = Worksheets.Add
    ws.Name = "打印机列表"
    
    ' 设置标题
    ws.Cells(1, 1).Value = "序号"
    ws.Cells(1, 2).Value = "打印机名称"
    ws.Cells(1, 3).Value = "是否为PDF打印机"
    
    ' 使用 WMI 获取所有打印机
    Set objWMIService = GetObject("winmgmts:\\.\root\cimv2")
    Set colPrinters = objWMIService.ExecQuery("Select * From Win32_Printer")
    
    i = 2
    For Each objPrinter In colPrinters
        ws.Cells(i, 1).Value = i - 1
        ws.Cells(i, 2).Value = objPrinter.Name
        
        ' 检查是否是PDF打印机
        If InStr(1, objPrinter.Name, "PDF", vbTextCompare) > 0 Or _
           InStr(1, objPrinter.Name, "Adobe", vbTextCompare) > 0 Or _
           InStr(1, objPrinter.Name, "虚拟", vbTextCompare) > 0 Or _
           InStr(1, objPrinter.Name, "Virtual", vbTextCompare) > 0 Then
            ws.Cells(i, 3).Value = "是"
        Else
            ws.Cells(i, 3).Value = "否"
        End If
        
        i = i + 1
    Next objPrinter
    
    ws.Columns("A:C").AutoFit
    
    MsgBox "已列出系统中所有打印机，共找到 " & (i - 2) & " 个打印机。", vbInformation
End Sub
