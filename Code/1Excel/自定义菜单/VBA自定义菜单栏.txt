'VBA 添加自定义菜单栏
Private Sub Workbook_Activate()
    On Error Resume Next
    ' 删除已有的"成绩管理"菜单栏（如果存在）
    MenuBars("成绩管理").Delete
    ' 建立一个自定义菜单栏"成绩管理"
    MenuBars.Add "成绩管理"
    ' 为"成绩管理"自定义菜单栏添加顶级菜单
    With MenuBars("成绩管理").Menus
        .Add "系统设置"
        .Add "学生信息管理"
        .Add "考勤管理"
        .Add "平时成绩管理"
        .Add "考试成绩管理"
        .Add "成绩查询与导出"
    End With
    ' 为"系统设置"菜单添加二级子菜单
    With MenuBars("成绩管理").Menus("系统设置").MenuItems
        .Add "数据库配置"
        .Add "用户权限设置"
        .Add "数据备份与恢复"
    End With
    ' 为"学生信息管理"菜单添加二级子菜单
    With MenuBars("成绩管理").Menus("学生信息管理").MenuItems
        .Add "添加新学生"
        .Add "编辑学生信息"
        .Add "删除学生信息"
    End With
    ' 激活"成绩管理"菜单栏
    MenuBars("成绩管理").Activate
End Sub
