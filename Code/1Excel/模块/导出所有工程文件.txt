Public Sub ExportCodeFile() '导出所有工程文件至
On Error GoTo ErrorHandler
    Dim fso As Object, Path$, Str$
    Dim vbComp As Object, i&, j&
    Path = ActiveWorkbook.Path '获取当前工作簿路径
    Str = "https://d.docs.live.net/" '替换字符
    If InStr(Path, Str) Then Path = Replace(Replace(Path, Str & Split(Path, "/")(3), Environ("OneDrive")), "/", "\") '若存在替换字符则将其替换为OneDrive字符
    With Application.FileDialog(msoFileDialogFolderPicker) '让用户选择导出文件夹
        .Title = "请选择目标文件夹" '标题文本
        .InitialFileName = Path & "\" '默认路径
        .AllowMultiSelect = False '不允许多选
        If .Show = 0 Then Exit Sub Else Path = .SelectedItems(1) & "\" '获取所选文件夹路径
    End With
    Set fso = CreateObject("Scripting.FileSystemObject") '创建文件系统对象
    Path = Path & "VBComponents From (" & ActiveWorkbook.Name & ") " & Format(Now, "yymmddhhmmss") '新文件夹路径
    If Not fso.FolderExists(Path) Then fso.CreateFolder Path '创建新文件夹
    Path = Path & "\" '新文件夹路径
    Application.EnableEvents = False '执行默认事件
    For Each vbComp In ActiveWorkbook.VBProject.VBComponents '遍历组件
        Select Case vbComp.Type '区分组件类型
            Case 1: vbComp.Export Path & vbComp.Name & ".bas" '标准模块
            Case 2: vbComp.Export Path & vbComp.Name & ".cls" '类模块
            Case 3: vbComp.Export Path & vbComp.Name & ".frm" '用户窗体
            Case 100: i = FreeFile: Str = Path & vbComp.Name & ".txt" 'Microsoft Excel对象（工作表、ThisWorkbook等）
                Open Str For Output As i '打开txt覆盖导入
                For j = 1 To vbComp.CodeModule.CountOfLines '遍历组件中的代码
                    Print #i, vbComp.CodeModule.Lines(j, 1) '写入txt
                Next j
                Close i '关闭txt
        End Select
    Next vbComp
ErrorHandler:
    If Err.Number Then '提示
        fso.deletefolder Left(Path, Len(Path) - 1), True
        MsgBox "导出失败：" & Err.Number & vbNewLine & Err.Description, vbCritical, "LELEDY" '错误提示
    Else
        If MsgBox("当前工作簿的所有工程文件已成功导出至文件夹：" & vbNewLine & Path & vbNewLine & vbNewLine & "是否需要打开文件夹查看？", vbInformation + vbOKCancel, "LELEDY") = vbOK Then '成功提示
            Shell "Explorer /select, " & Path, vbNormalFocus '打开文件夹
        End If
    End If
    Set fso = Nothing: Set vbComp = Nothing '清理
    Application.EnableEvents = True '用户选择执行
End Sub
