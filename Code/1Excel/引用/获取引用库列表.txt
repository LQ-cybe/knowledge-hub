'获取当前文档所有引用的详细信息
Sub Grab_References()
    Dim n As Integer
    Cells.Clear
    range("A1:H1") = Array("名称", "全名", "全球唯一标识符", "主版本号", "次版本号", "库文件路径", "是否内置", "是否丢失")
    n = 2
    For Each ref In ThisWorkbook.VBProject.References
        Cells(n, 1) = ref.name
        Cells(n, 2) = ref.Description
        Cells(n, 3) = ref.GUID
        Cells(n, 4) = ref.Major
        Cells(n, 5) = ref.Minor
        Cells(n, 6) = ref.fullPath
        Cells(n, 7) = ref.BuiltIn
        Cells(n, 8) = ref.IsBroken
        n = n + 1
    Next
End Sub
