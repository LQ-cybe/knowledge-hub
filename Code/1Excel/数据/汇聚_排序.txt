' 使用字典顺序对整个数据范围进行排序
Sub SortEntireRangeWithDictionary(rng As Range, orderDict As Object)
    Dim ws As Worksheet
    Set ws = rng.Worksheet

    ' 创建临时工作表用于排序
    Dim tempWs As Worksheet
    Set tempWs = ThisWorkbook.Worksheets.Add

    ' 复制整个数据范围到临时工作表
    rng.Copy tempWs.Range("A1")

    ' 获取临时工作表中的数据范围
    Dim lastRow As Long
    Dim lastCol As Long
    With tempWs
        lastRow = .Cells(.Rows.Count, 1).End(xlUp).Row
        lastCol = .Cells(1, .Columns.Count).End(xlToLeft).Column
    End With

    ' 在临时工作表的最后一列之后添加排序辅助列
    Dim sortCol As Long
    sortCol = lastCol + 1

    ' 为每一行添加排序值
    Dim i As Long
    For i = 1 To lastRow
        Dim cellValue As String
        cellValue = CStr(tempWs.Cells(i, 1).Value)    ' 假设第一列是排序依据

        If orderDict.Exists(cellValue) Then
            tempWs.Cells(i, sortCol).Value = orderDict(cellValue)
        Else
            tempWs.Cells(i, sortCol).Value = 999999    ' 不在字典中的项放在最后
        End If
    Next i

    ' 按辅助列排序整个数据范围
    With tempWs.Sort
        .SortFields.Clear
        .SortFields.Add Key:=tempWs.Cells(1, sortCol), Order:=xlAscending
        .SetRange tempWs.Range(tempWs.Cells(1, 1), tempWs.Cells(lastRow, sortCol))
        .Header = xlNo    ' 根据实际情况调整，xlYes含标题
        .Apply
    End With

    ' 复制排序后的数据（不包括辅助列）回原始工作表
    tempWs.Range(tempWs.Cells(1, 1), tempWs.Cells(lastRow, lastCol)).Copy
    rng.PasteSpecial xlPasteValues

    ' 删除临时工作表
    Application.DisplayAlerts = False
    tempWs.Delete
    Application.DisplayAlerts = True

    ' 清除剪贴板
    Application.CutCopyMode = False
End Sub
