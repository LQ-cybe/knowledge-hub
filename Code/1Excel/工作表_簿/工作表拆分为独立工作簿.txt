'工作表拆分为独立工作簿（用户自选保存文件夹）
Sub WorksheetsSplitSeparateWorkbooks()
    Dim oWK As Worksheet
    Dim oWB As Workbook
    Dim sPath As String
    Dim sName As String
    Dim fd As FileDialog
    If MsgBox("现在开始将把各工作表独立另存为工作簿文件，请再次检查格式数据是否正确？", vbYesNo, "重要提示") = vbNo Then
        Exit Sub
    End If
    Set fd = Application.FileDialog(msoFileDialogFolderPicker)
    With fd
        .Title = "请选择保存文件夹"
        If .Show <> -1 Then
            MsgBox "未选择文件夹，程序退出！", vbExclamation
            Exit Sub
        End If
        sPath = .SelectedItems(1)
    End With
    Set fd = Nothing
    Application.ScreenUpdating = False
    Application.Calculation = xlCalculationManual
    Application.DisplayAlerts = False
    For Each oWK In ThisWorkbook.Worksheets
        With oWK
            sName = .Name
            .Copy                           ' 复制工作表到新工作簿
            Set oWB = ActiveWorkbook        ' 指向新工作簿
            oWB.SaveAs sPath & "\" & sName, xlOpenXMLWorkbook  ' 保存
            oWB.Close                       ' 关闭新工作簿
        End With
    Next
    Application.ScreenUpdating = True
    Application.Calculation = xlCalculationAutomatic
    Application.DisplayAlerts = True

    MsgBox " 所有工作表已拆分完成！" & vbCrLf & "保存路径：" & sPath, vbInformation
End Sub
