    ' 64位API声明（带PtrSafe）
    Declare PtrSafe Function OpenClipboard Lib "user32" (ByVal hwnd As LongPtr) As Long
    Declare PtrSafe Function EmptyClipboard Lib "user32" () As Long
    Declare PtrSafe Function SetClipboardData Lib "user32" (ByVal wFormat As Long, ByVal hMem As LongPtr) As LongPtr
    Declare PtrSafe Function CloseClipboard Lib "user32" () As Long
    Declare PtrSafe Function GlobalAlloc Lib "kernel32" (ByVal wFlags As Long, ByVal dwBytes As LongPtr) As LongPtr
    Declare PtrSafe Function GlobalLock Lib "kernel32" (ByVal hMem As LongPtr) As LongPtr
    Declare PtrSafe Function GlobalUnlock Lib "kernel32" (ByVal hMem As LongPtr) As Long
    Declare PtrSafe Sub CopyMemory Lib "kernel32" Alias "RtlMoveMemory" (ByVal Destination As LongPtr, ByRef Source As Any, ByVal Length As LongPtr)
'复制指定第一个的内容到剪切板
Sub CopyD13ToClipboard_x64()
    Dim cellValue As String
    Dim hGlobal As LongPtr
    Dim lpGlobal As LongPtr
    Dim byteData() As Byte  ' 用于精确控制字节数据
    Dim i As Long

    
    ' 常量定义
    Const CF_TEXT = 1
    Const GMEM_MOVEABLE = &H2
    Const GMEM_ZEROINIT = &H40

    ' 获取D13单元格值并处理首尾双引号
    On Error Resume Next
    cellValue = CStr(ActiveSheet.Range("D13").Value)
    On Error GoTo 0

    ' 移除首尾双引号
    If Len(cellValue) > 0 Then
        If Left(cellValue, 1) = """" Then cellValue = Mid(cellValue, 2)
        If Right(cellValue, 1) = """" Then cellValue = Left(cellValue, Len(cellValue) - 1)
    End If

    ' 将字符串转换为ANSI字节数组（含终止符\0）
    byteData = StrConv(cellValue & Chr(0), vbFromUnicode)  ' 手动添加null终止符

    ' 复制到剪贴板
    If OpenClipboard(0) Then
        EmptyClipboard

        ' 按字节数组长度分配内存（确保包含终止符）
        hGlobal = GlobalAlloc(GMEM_MOVEABLE Or GMEM_ZEROINIT, UBound(byteData) + 1)
        If hGlobal <> 0 Then
            lpGlobal = GlobalLock(hGlobal)
            If lpGlobal <> 0 Then
                ' 精确复制字节数组到内存（避免垃圾数据）
                CopyMemory lpGlobal, byteData(0), UBound(byteData) + 1
                GlobalUnlock hGlobal
                SetClipboardData CF_TEXT, hGlobal
            End If
        End If

        CloseClipboard
    End If

'    MsgBox "已复制D13内容到剪贴板（无多余字符）", vbInformation
End Sub
