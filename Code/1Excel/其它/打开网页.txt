Public Declare PtrSafe Function ShellExecute Lib "shell32.dll" Alias "ShellExecuteA" (ByVal hwnd As Long, ByVal lpOperation As String, ByVal lpFile As String, ByVal lpParameters As String, ByVal lpDirectory As String, ByVal nShowCmd As Long) As Long
Public Const SW_SHOWNORMAL = 1
'API打开网页
Sub ApiOpenWebPage()
    ShellExecute 0, "open", "http://www.exceloffice.net/", vbNullString, vbNullString, SW_SHOWNORMAL
End Sub

'打开网页
Sub OpenAWebPage()
Excel.Application.ActiveWorkbook.FollowHyperlink "http://www.exceloffice.net/"
End Sub

'指定浏览器打开网页
Sub SpecifyTheBrowserToOpenTheWebPage()
    Dim sUrl As String
    sUrl = "http://www.exceloffice.net/"
    Shell "C:\Program Files (x86)\Microsoft\Edge\Application\msedge.exe " & sUrl, vbNormalFocus
End Sub
