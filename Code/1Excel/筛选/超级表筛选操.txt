Sub 筛选()
    Dim SXname As String
    SXname = Range("K1").Value & ""
    ActiveSheet.ListObjects("表1").Range.AutoFilter Field:=1, Criteria1:="*" & SXname & "*"
End Sub
Sub 清除筛选()
    Dim ws As Worksheet
    Dim lo As ListObject
    Dim colCount As Integer
    
    Set ws = ThisWorkbook.ActiveSheet
    Set lo = ws.ListObjects("表1")
    
    colCount = lo.ListColumns.Count
    For I = 1 To colCount
        ActiveSheet.ListObjects("表1").Range.AutoFilter Field:=I
    Next
End Sub
'切换视图，定位到筛选后的数据
Sub ScrollDownFromA1()
    Dim ws As Worksheet
    Set ws = ThisWorkbook.ActiveSheet
    Dim targetCell As Range
    '定位到A1
    Set targetCell = ws.Range("A1")
    targetCell.Activate
    '下移一行
    Set targetCell = targetCell.Offset(1, 0)
    targetCell.Activate
    '滚动窗口使目标单元格可见
    Application.Goto targetCell, True
End Sub
