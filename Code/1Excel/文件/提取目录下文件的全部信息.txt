    '提取文件夹下的所有Excel文件的全部信息
Sub GetWorkbookInfos(folderPath As String)
    Dim objWorkSheet As Worksheet
    Dim objWorkbook As Workbook
    Dim filePath As String
    Dim prop As Object
    Dim r As Integer
    Dim c As Integer
    Call RunInBackground(True)
    r = 1
    On Error Resume Next
    Set objWorkSheet = ActiveSheet
    If Right(folderPath, 1) <> "\" Then
        folderPath = folderPath & "\"
    End If
    filePath = Dir(folderPath & "*.xl*")
    Do While filePath <> ""
        Set objWorkbook = Workbooks.Open(Filename:=folderPath & filePath, UpdateLinks:=0)
        c = 1
        If r = 1 Then
            objWorkSheet.Cells(r, c).Value = "文件大小(KB)"
            For Each prop In objWorkbook.BuiltinDocumentProperties
                objWorkSheet.Cells(r, c + 1).Value = prop.Name
                c = c + 1
            Next
            r = r + 1
        End If
        c = 1
        objWorkSheet.Cells(r, c).Value = FileLen(objWorkbook.FullName) / 1024
        For Each prop In objWorkbook.BuiltinDocumentProperties
            c = c + 1
            objWorkSheet.Cells(r, c).Value = IIf(IsError(prop.Value), "", prop.Value)
        Next
        objWorkbook.Close SaveChanges:=False
        filePath = Dir
        r = r + 1
    Loop
    On Error GoTo 0
    Call RunInBackground(False)
End Sub
    '后台运行子过程
Sub RunInBackground(isBackground As Boolean)
    If isBackground Then
        Application.ScreenUpdating = False
        Application.CutCopyMode = False
        Application.DisplayAlerts = False
        Application.EnableEvents = False
        Application.WindowState = xlMinimized                  '最小化Excel窗口
        Application.Calculation = xlCalculationManual
    Else
        Application.ScreenUpdating = True
        Application.CutCopyMode = True
        Application.DisplayAlerts = True
        Application.EnableEvents = True
        AppActivate Application.Caption
        Application.WindowState = xlMaximized                  '最大化Excel窗口
        Application.Calculation = xlCalculationAutomatic
    End If
End Sub
Sub GetWorkbookInfosTest()
    GetWorkbookInfos "D:\Office办公\Excel文件"
End Sub
