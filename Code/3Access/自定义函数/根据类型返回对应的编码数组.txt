    Dim Parameter As Integer
'    自定义函数，提取类别对应的月编码以数组的形式返回
Function GetArrayByTableNameAndName(tableName As String, arrayName As String) As Variant
    Dim db As DAO.Database
    Dim rs As DAO.Recordset
    Dim resultArray() As Variant
    Dim i As Integer
    Dim found As Boolean
    
    Set db = CurrentDb
    Set rs = db.OpenRecordset(tableName, dbOpenDynaset)
    
    found = False
    Do While Not rs.EOF
        If rs.Fields("类别名称") = arrayName Then
            ReDim resultArray(2 To Parameter)
            For i = 2 To Parameter
                resultArray(i) = rs.Fields(i)
            Next i
            found = True
            Exit Do
        End If
        rs.MoveNext
    Loop
    
    rs.Close
    Set rs = Nothing
    Set db = Nothing
    
    If found Then
        GetArrayByTableNameAndName = resultArray
    Else
        GetArrayByTableNameAndName = Empty
    End If
End Function
