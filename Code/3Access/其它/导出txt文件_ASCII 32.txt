'列之间使用的是普通空格（ASCII 32）分隔
Sub ExportTableToTxt()                                     '导出表数据到到桌面TXT文件
    Dim desktopPath As String
    Dim txtFilePath As String
    Dim db As DAO.Database
    Dim rs As DAO.Recordset
    Dim fso As Object
    Dim txtFile As Object
    Dim fieldCount As Integer
    Dim I As Integer
    Dim line As String
    Dim startTime As Double
    Dim endTime As Double
    startTime = Timer
    desktopPath = CreateObject("WScript.Shell").SpecialFolders("Desktop")    ' 获取桌面路径
    ' 构建 TXT 文件的完整路径
    txtFilePath = desktopPath & "\" & nameTxt & ".txt"
    On Error Resume Next
    ' 删除已存在的同名 TXT 文件
    Kill txtFilePath
    On Error GoTo 0
    ' 打开数据库和记录集
    Set db = CurrentDb
    Set rs = db.OpenRecordset(Serialnumberdata)
    Set db = CurrentDb
    '假设表中有A字段，按A字段降序,DESC=降序，ASC=升序
    Set rs = db.OpenRecordset("SELECT * FROM " & Serialnumberdata & " ORDER BY A " & sequence & " ")
    ' 创建文件系统对象
    Set fso = CreateObject("Scripting.FileSystemObject")
    Set txtFile = fso.CreateTextFile(txtFilePath, True)
    ' 获取字段数量
    fieldCount = rs.Fields.count
    ' 写入字段名
    For I = 0 To fieldCount - 1
        line = line & Replace(Replace(rs.Fields(I).Name, """", ""), ",", "")
        If I < fieldCount - 1 Then
            line = line & " "
        End If
    Next I
    txtFile.WriteLine line
    line = ""
    ' 写入记录数据
    '    rs.MoveFirst
    Do While Not rs.EOF
        For I = 0 To fieldCount - 1
            line = line & Replace(Replace(rs.Fields(I).value, """", ""), ",", "")
            If I < fieldCount - 1 Then
                line = line & " "
            End If
        Next I
        txtFile.WriteLine line
        line = ""
        rs.MoveNext
    Loop
    ' 关闭对象
    rs.Close
    txtFile.Close
    Set rs = Nothing
    Set txtFile = Nothing
    Set fso = Nothing
    Set db = Nothing
    '    MsgBox "数据已成功导出到 " & txtFilePath
    endTime = Timer
    '        MsgBox "操作完成，数据已成功导出到" & txtFilePath & "运行时间:" & Format((endTime - startTime), "000") & "秒"
End Sub
