'备份文件夹下的文件到目标文件夹下，不含子文件夹和文件
Sub CopyFolderUsingFileCopy(sourceFolder As String, destinationFolder As String)
    Dim file As Integer
    Dim sourceFile As String
    Dim destinationFile As String
    
   '确保目标文件夹存在，如果不存在则创建
    If Dir(destinationFolder, vbDirectory) = "" Then
        MkDir destinationFolder
    End If
    
    sourceFile = Dir(sourceFolder & "\*.*")
    Do While sourceFile <> ""
        If (GetAttr(sourceFolder & "\" & sourceFile) And vbDirectory) = 0 Then '检查是否为文件
            destinationFile = destinationFolder & "\" & sourceFile
            FileCopy sourceFolder & "\" & sourceFile, destinationFile
        End If
        sourceFile = Dir
    Loop
End Sub
