'备份文件夹下的文件到目标文件夹下，含子文件夹和文件
Sub CopyFolderUsingFSO(sourceFolder As String, destinationFolder As String)
    Dim fso As Object
    Dim source As Object
    Dim destination As Object
    
   '创建FileSystemObject对象
    Set fso = CreateObject("Scripting.FileSystemObject")
    
   '确保目标文件夹存在，如果不存在则创建
    If Not fso.FolderExists(destinationFolder) Then
        fso.CreateFolder destinationFolder
    End If
    
   '获取源文件夹对象
    Set source = fso.GetFolder(sourceFolder)
    
   '遍历源文件夹中的所有文件和子文件夹
    For Each file In source.Files
        file.Copy destinationFolder & "\" & file.Name
    Next file
    
    For Each subFolder In source.SubFolders
        CopyFolderUsingFSO subFolder.Path, destinationFolder & "\" & subFolder.Name
    Next subFolder
    
   '释放对象
    Set destination = Nothing
    Set source = Nothing
    Set fso = Nothing
End Sub
