'创建文件夹
Private Sub Command0_Click()
    Dim folderPath As String
    folderPath = "C:\TestFolder" '指定要创建的文件夹路径，可按需修改
    On Error Resume Next '忽略错误，若文件夹已存在则不报错
    MkDir folderPath
    On Error GoTo 0 '恢复错误处理
    If Err.Number = 0 Then
        MsgBox "文件夹创建成功！"
    Else
        MsgBox "文件夹创建失败，可能已存在或路径有误。"
    End If
End Sub
