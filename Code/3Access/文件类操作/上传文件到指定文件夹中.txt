    Dim target As Variant
    Public Foldername As String
    Dim folderPath As String                                   '定义完整的软件设置变量
    '调用软件设置中文件夹地址
Sub GetAndShow()
    Dim db As DAO.Database
    Dim rs As DAO.Recordset
    Dim i As Integer
    Set db = CurrentDb()
    Set rs = db.OpenRecordset("Sys_软件设置", dbOpenDynaset)
    For i = 1 To 0
        If Not rs.EOF Then
            rs.MoveNext
        End If
    Next i
    If Not rs.EOF Then
        target = rs.Fields("文件夹地址").value
        '        Createfolder        '调用创建文件夹过程
    Else
        MsgBox "未找到软件设置，请检查软件设置"
    End If
    rs.Close
    Set rs = Nothing
    Set db = Nothing
End Sub
    '创建文件夹过程
Sub Createfolder()
    'MsgBox Foldername
    Call GetAndShow                                            '调用设置文件夹地址的子过程
    '    Dim Foldername As String
    '    Foldername = "测试文件夹"
    folderPath = target & "\" & Foldername                     '指定要创建的文件夹路径，可按需修改
    On Error Resume Next                                       '忽略错误，若文件夹已存在则不报错
    MkDir folderPath
    On Error GoTo 0                                            '恢复错误处理
    If Err.Number = 0 Then
        '        MsgBox "文件夹创建成功！" & folderPath
    Else
        MsgBox "文件夹创建失败，可能已存在或路径有误。"
    End If
    Call OpenFileDialogAndCopy                                 '调用选择文件及复制过程
End Sub
    '打开文件选择窗口选择文件并复制软件设置的目录下
Sub OpenFileDialogAndCopy()
    Dim fileDialog As fileDialog
    Dim selectedFile As Variant
    Dim destinationFolder As String
    Dim result As Integer
    ' 创建文件对话框对象
    Set fileDialog = Application.fileDialog(msoFileDialogFilePicker)
    ' 设置对话框标题
    fileDialog.Title = "请选择文件"
    ' 设置是否允许多选
    fileDialog.AllowMultiSelect = True
    ' 设置文件筛选器
    fileDialog.Filters.Clear
    fileDialog.Filters.Add "所有文件", "*.*"
    '    fileDialog.Filters.Add "文本文件", "*.txt"
    ' 显示文件对话框
    If fileDialog.Show = -1 Then
        ' 获取用户选择的文件路径
        selectedFile = fileDialog.SelectedItems(1)
        ' 指定目标文件夹路径，可以是共享盘路径
        destinationFolder = folderPath
        '        MsgBox folderPath
        '        ' 检查目标文件夹是否存在，如果不存在则创建
        '        If Dir(destinationFolder, vbDirectory) = "" Then
        '            MkDir destinationFolder
        '        End If
        ' 复制文件到目标文件夹
        On Error Resume Next
        FileCopy selectedFile, destinationFolder & "\" & Dir(selectedFile)
        result = Err.Number
        On Error GoTo 0
        If result = 0 Then
            MsgBox "文件上传成功！"
        Else
            MsgBox "文件上传失败，错误代码：" & result
        End If
    Else
        MsgBox "你没有选择任何文件。"
    End If
    ' 释放对象
    Set fileDialog = Nothing
End Sub
