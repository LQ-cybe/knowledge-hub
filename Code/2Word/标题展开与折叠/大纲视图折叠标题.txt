'在大纲视图下折叠显示指定层级的标题
Sub ExpandCollapseByLevel()
    Dim level As Integer
    Dim doc As Document
    Dim wdWindow As Window

    ' 获取当前活动文档
    On Error Resume Next
    Set doc = ActiveDocument
    If doc Is Nothing Then
        MsgBox "没有打开的文档！", vbExclamation
        Exit Sub
    End If

    ' 获取活动窗口
    Set wdWindow = doc.ActiveWindow
    If wdWindow Is Nothing Then
        MsgBox "无法获取文档窗口！", vbExclamation
        Exit Sub
    End If
    On Error GoTo 0

    ' 获取用户输入
    level = InputBox("请输入要显示的标题级别（1-9）：" & vbCrLf & _
                     "1 = 只显示1级标题" & vbCrLf & _
                     "2 = 显示1-2级标题" & vbCrLf & _
                     "...（以此类推）" & vbCrLf & _
                     "9 = 显示所有标题级别", _
                     "设置标题显示级别", 9)

    ' 检查用户是否取消了输入
    If level = 0 Then
        Exit Sub
    End If

    If level >= 1 And level <= 9 Then
        Application.ScreenUpdating = False

        ' 确保有活动文档
        If Not doc Is Nothing Then
            With wdWindow.View
                If .Type <> wdOutlineView Then
                    .Type = wdOutlineView
                End If
                .ShowHeading level
            End With
        End If

        Application.ScreenUpdating = True
'        MsgBox "已设置显示" & IIf(level = 9, "所有", level & "级及以下") & "标题", vbInformation
    ElseIf level <> 0 Then    ' 排除用户点击取消的情况
        MsgBox "请输入1-9之间的数字", vbExclamation
    End If
End Sub
