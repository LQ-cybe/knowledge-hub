    With ActiveWindow
        .DisplayHorizontalScrollBar = False             '隐藏水平滚动条
        .DisplayVerticalScrollBar = False                  '隐藏垂直滚动条
        .DisplayWorkbookTabs = False                     '隐藏工作表标签
        .DisplayGridlines = False                                '隐藏网格线
        .DisplayHeadings = False                            '隐藏标题栏
	ActiveSheet.DisplayPageBreaks = False		'隐藏分页符和网格线
    End With

Sub a()
    AAA = True
    Application.DisplayAlerts = False                          ' 关闭提示框
    Application.ScreenUpdating = False                         ' 关闭屏幕更新
    Application.AskToUpdateLinks = False                       ' 关闭自动询问是否更新链接的功能
    ActiveWindow.DisplayHorizontalScrollBar = AAA              '隐藏水平滚动条
    ActiveWindow.DisplayVerticalScrollBar = AAA                '隐藏垂直滚动条
    Application.DisplayFormulaBar = AAA                        '隐藏编辑栏
    ActiveWindow.DisplayHeadings = AAA                         '隐藏标题栏
    ActiveWindow.DisplayGridlines = AAA                        '隐藏网格线
    Application.DisplayStatusBar = AAA                         '隐藏状态栏
    
    ActiveSheet.ScrollArea = "$A$1:$F$20"                      '将工作表的滚动区域限制在A1:F20
    ActiveSheet.ScrollArea = ""                                             '取消工作表的滚动限制
End Sub
