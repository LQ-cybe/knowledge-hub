'判断某文件夹是否存在，如果存在就打开，如果不存在则创建并打开
Sub CheckAndOperateFolder()
    Dim folderPath As String
    folderPath = "D:\LQ\W020000294"
    '请将此处路径替换为实际想要操作的文件夹路径，例如 "C:\TestFolder"
    Dim fso As Object
    Set fso = CreateObject("Scripting.FileSystemObject")
    If fso.FolderExists(folderPath) Then
        '如果文件夹存在，打开该文件夹
        Call Shell("explorer.exe " & folderPath, vbNormalFocus)
    Else
        '如果文件夹不存在，创建该文件夹
        fso.CreateFolder folderPath
        '创建后打开该文件夹
        Call Shell("explorer.exe " & folderPath, vbNormalFocus)
    End If
    Set fso = Nothing
End Sub
